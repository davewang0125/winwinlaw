⚙️ SETUP (5 minutes)
--------------------
1. Install dependencies:
   cd profiles && npm install

2. Setup database:
   createdb winwinlaw
   cp .env.example .env
   # Edit .env with your DB credentials
   npm run db:init

3. Verify:
   psql winwinlaw -c "SELECT COUNT(*) FROM practice_areas;"

📊 DATABASE QUERIES
-------------------
# Search lawyers
SELECT * FROM legal_professionals
WHERE full_name ILIKE '%smith%'
AND is_active = true;

# With practice areas
SELECT * FROM active_professionals_with_practice_areas
WHERE practice_areas LIKE '%litigation%';

# Recent additions
SELECT full_name, city, created_at
FROM legal_professionals
WHERE created_at > NOW() - INTERVAL '7 days';

# Stats by source
SELECT source, COUNT(*)
FROM legal_professionals
GROUP BY source;

🔍 SEARCH API
-------------
# Create endpoint: app/api/search/lawyers/route.ts
import { searchProfessionals } from '@/profiles/database/operations'

export async function GET(request: Request) {
  const { searchParams } = new URL(request.url)
  const query = searchParams.get('q') || ''
  const results = await searchProfessionals(query)
  return Response.json({ results })
}

# Use in landing2:
const results = await fetch(`/api/search/lawyers?q=${query}`)
